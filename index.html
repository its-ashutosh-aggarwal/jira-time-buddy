<script>

const calls = [

 {id:1,employee:"Ashutosh Aggarwal",client:"Client Alpha",meeting:"Production support discussion",time:"10:15 AM",duration:"27m",issue:"CLIENT-123",status:"Pending"},

 {id:2,employee:"Rahul Sharma",client:"Client Beta",meeting:"Requirement clarification",time:"11:05 AM",duration:"33m",issue:"SUPPORT-456",status:"Logged"},

 {id:3,employee:"Priya Singh",client:"Client Gamma",meeting:"Technical clarification",time:"12:20 PM",duration:"18m",issue:"",status:"Pending"},

 {id:4,employee:"Ashutosh Aggarwal",client:"Client Delta",meeting:"Deployment follow-up",time:"1:10 PM",duration:"42m",issue:"PROJ-789",status:"Logged"},

 {id:5,employee:"Rahul Sharma",client:"Client Alpha",meeting:"Incident review",time:"2:30 PM",duration:"16m",issue:"",status:"Pending"},

 {id:6,employee:"Priya Singh",client:"Client Beta",meeting:"Integration discussion",time:"3:05 PM",duration:"24m",issue:"SUPPORT-456",status:"Logged"}

];


let history = [];

let selected = null;

let role = "manager";


function statusBadge(s) {

 return `<span class="pill ${s==="Logged"?"logged":"pending"}">${s}</span>`;

}


function navigate(page) {

 document.querySelectorAll(".section").forEach(x=>x.classList.remove("active"));

 document.getElementById(page).classList.add("active");


 document.querySelectorAll(".nav").forEach(n=>{

  n.classList.toggle("active",n.dataset.page===page);

 });


 const titles = {

  overview:"Overview",

  calls:"Call inbox",

  team:"Team monitoring",

  history:"Worklog history",

  about:"About & integrations"

 };


 document.getElementById("pageTitle").textContent=titles[page];

 render();

}


document.querySelectorAll(".nav").forEach(n=>{

 n.addEventListener("click",()=>navigate(n.dataset.page));

});


document.getElementById("role").addEventListener("change",e=>{

 role=e.target.value;

 document.getElementById("greeting").textContent=

  role==="manager"?"Team overview":"My worklog overview";

 render();

});


function openModal(id) {

 selected=calls.find(c=>c.id===id);

 if(!selected)return;


 document.getElementById("modalCall").textContent=

  selected.client+" · "+selected.meeting+" · "+selected.duration;


 document.getElementById("issue").value=selected.issue;

 document.getElementById("duration").value=selected.duration;

 document.getElementById("modal").classList.add("open");

}


function closeModal() {

 document.getElementById("modal").classList.remove("open");

}


document.getElementById("worklogForm").addEventListener("submit",e=>{

 e.preventDefault();

 if(!selected)return;


 if(selected.status==="Logged") {

  alert("This call is already logged.");

  return;

 }


 const issue=document.getElementById("issue").value.trim().toUpperCase();


 if(!/^[A-Z][A-Z0-9_]*-\d+$/.test(issue)) {

  alert("Enter a valid Jira issue key, e.g. CLIENT-123.");

  return;

 }


 const entry={

  ...selected,

  issue,

  duration:document.getElementById("duration").value,

  description:document.getElementById("description").value,

  status:"Logged"

 };


 selected.status="Logged";

 selected.issue=issue;

 selected.duration=entry.duration;

 history.unshift(entry);


 closeModal();

 render();

 navigate("history");


 alert("Demo worklog saved in this browser session only. Jira was not contacted.");

});


function render() {

 const visible=role==="manager"

  ?calls

  :calls.filter(c=>c.employee==="Ashutosh Aggarwal");


 const logged=visible.filter(c=>c.status==="Logged").length;

 const pending=visible.length-logged;


 const total=visible.reduce((n,c)=>n+parseInt(c.duration),0);


 const metrics=[

  ["Client calls",visible.length,"Sample detected calls"],

  ["Time captured",total+"m","Total sample duration"],

  ["Pending worklogs",pending,"Awaiting review"],

  ["Logged worklogs",logged,"Completed sample entries"]

 ];


 document.getElementById("metrics").innerHTML=metrics.map(m=>`

  <div class="metric">

   <div class="metric-title">${m[0]}</div>

   <div class="metric-value">${m[1]}</div>

   <div class="metric-note">${m[2]}</div>

  </div>

 `).join("");


 document.getElementById("recentRows").innerHTML=

  visible.slice(0,5).map(c=>`

   <tr>

    <td>${c.employee}</td>

    <td>${c.client}<br>${c.meeting}</td>

    <td>${c.duration}</td>

    <td>${c.issue||"—"}</td>

    <td>${statusBadge(c.status)}</td>

   </tr>

  `).join("");


 const callSearch=(document.getElementById("callSearch")?.value||"").toLowerCase();


 document.getElementById("callCards").innerHTML=

  visible.filter(c=>(c.client+" "+c.meeting).toLowerCase().includes(callSearch))

  .map(c=>`

   <div class="callcard">

    <h3>${c.client}</h3>

    <p>${c.meeting}</p>

    <p>${c.employee} · ${c.time}</p>

    <p><strong>${c.duration}</strong> · ${c.issue||"No Jira issue assigned"}</p>

    ${statusBadge(c.status)}

    <p>

     <button class="action" onclick="openModal(${c.id})"

      ${c.status==="Logged"?"disabled":""}>

      ${c.status==="Logged"?"Already logged":"Review worklog →"}

     </button>

    </p>

   </div>

  `).join("")||"<p>No matching calls.</p>";


 const search=(document.getElementById("teamSearch")?.value||"").toLowerCase();

 const filter=document.getElementById("statusFilter")?.value||"all";


 document.getElementById("teamRows").innerHTML=

  visible.filter(c=>

   (c.employee+" "+c.client+" "+c.meeting+" "+c.issue).toLowerCase().includes(search)

   &&(filter==="all"||c.status===filter)

  ).map(c=>`

   <tr>

    <td>${c.employee}</td>

    <td>${c.client}<br>${c.meeting}</td>

    <td>${c.duration}</td>

    <td>${c.issue||"—"}</td>

    <td>${statusBadge(c.status)}</td>

    <td>${c.status==="Pending"

     ?`<button class="action" onclick="openModal(${c.id})">Review</button>`

     :"—"}

    </td>

   </tr>

  `).join("");


 document.querySelector('[data-page="team"]').style.display=

  role==="manager"?"block":"none";


 const hist=role==="manager"

  ?history

  :history.filter(c=>c.employee==="Ashutosh Aggarwal");


 document.getElementById("historyRows").innerHTML=

  hist.map(c=>`

   <tr>

    <td>${c.employee}</td>

    <td>${c.client}</td>

    <td>${c.issue}</td>

    <td>${c.duration}</td>

    <td>${c.description}</td>

   </tr>

  `).join("")||

  '<tr><td colspan="5">No worklogs submitted in this demo session.</td></tr>';

}


document.getElementById("callSearch").addEventListener("input",render);

document.getElementById("teamSearch").addEventListener("input",render);

document.getElementById("statusFilter").addEventListener("change",render);


render();

</script>
